def routes(p,b,goal,walkable,successors):
    queue,seen,out = [(p,None,0)],{p},[]
    for i,(x,f,k) in enumerate(queue):
        if i==53: return out,i,True
        for a,(q,n) in successors((x,b)).items():
            s = a if f is None else f
            if n==b and q not in seen: seen.add(q); queue.append((q,s,k+1))
            if n!=b: r,c = n[0]-b[0],n[1]-b[1]; v = n==goal or walkable((n[0]+r,n[1]+c)) or (walkable((n[0]+c,n[1]-r)) and walkable((n[0]-c,n[1]+r))); out.append((s,q,n,k,v))
    return out,len(queue),False
def dist(x,y): return abs(x[0]-y[0])+abs(x[1]-y[1])
def plan(state, history, goal, walkable, successors, base_search, rng):

            if not direct and openings: break
